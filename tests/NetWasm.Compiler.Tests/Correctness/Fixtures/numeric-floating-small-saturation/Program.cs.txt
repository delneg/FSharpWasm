using System;

namespace NetWasm.Correctness.FloatingSmallSaturation;

public static class EntryPoint
{
    // Unchecked float and double conversions to sbyte, byte, short, ushort and
    // char. .NET 11 saturates them at the destination's bounds and converts NaN
    // to 0; earlier runtimes wrapped out-of-range values.
    private static readonly double[] Values =
    [
        double.NaN, double.PositiveInfinity, double.NegativeInfinity, double.MaxValue, double.MinValue,
        1e30, -1e30, 4294967296.0, 2147483648.0, -2147483649.0, 1e10, -1e10,
        65536.0, 65535.9, 32768.0, 32767.9, -32768.9, -32769.0, 300.7, -200.5,
        256.0, 255.9, 128.0, 127.9, -128.9, -129.0, 0.99, -0.5, -1.0, -0.0, double.Epsilon,
    ];

    public static int Run(int input)
    {
        var value = Values[input % Values.Length];
        var destination = input / Values.Length;
        return destination < 5 ? FromDouble(value, destination) : FromSingle((float)value, destination - 5);
    }

    private static int FromDouble(double value, int destination) => destination switch
    {
        0 => unchecked((sbyte)value),
        1 => unchecked((byte)value),
        2 => unchecked((short)value),
        3 => unchecked((ushort)value),
        _ => unchecked((char)value),
    };

    private static int FromSingle(float value, int destination) => destination switch
    {
        0 => unchecked((sbyte)value),
        1 => unchecked((byte)value),
        2 => unchecked((short)value),
        3 => unchecked((ushort)value),
        _ => unchecked((char)value),
    };
