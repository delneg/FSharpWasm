using System;
using System.Reflection;
using System.Runtime.CompilerServices;

namespace NetWasm.Correctness.UnwindRoots;

public sealed class Payload(int number)
{
    public int Number = number;
}

public sealed class PayloadException(Payload value) : Exception
{
    public Payload Value = value;
}

public sealed class Marker : Exception;

public sealed class Resource : IDisposable
{
    private readonly Payload _value = new(41);
    public void Dispose()
    {
        EntryPoint.Mark(2);
        GC.Collect();
        _value.Number++;
        EntryPoint.LastDisposed = _value;
        EntryPoint.Mark(3);
    }
}

public struct Block
{
    public long A, B, C, D, E, F, G, H;
}

public static class EntryPoint
{
    private static int[] _events = [];
    private static int _count;
    public static Payload? LastDisposed;
    private static WeakReference<Payload>? _held;
    private static int _sink;

    // An exception leaves every frame between its throw and its catch. Whatever the
    // frames held must become unreachable, and the frames entered afterwards must
    // behave as if the skipped ones had returned.
    public static int Rewind(int input)
    {
        _held = null;
        switch (input)
        {
            case 0:
                try { Hold(12); return -1; }
                catch (Marker) { Collect(); }
                return _held!.TryGetTarget(out _) ? -2 : 42;
            case 1:
            {
                var outer = new Block { A = 40, H = 2 };
                Touch(ref outer);
                for (var round = 0; round < 3; round++)
                {
                    try { Fill(100); return -3; }
                    catch (Marker) { Collect(); }
                }
                var total = Sum(100);
                return outer.A + outer.H == 42 && outer.B == 41 && total == 15150 ? 42 : -4;
            }
            case 2:
            {
                Func<int, int> hold = HoldAndThrow;
                try { hold(6); return -5; }
                catch (Marker) { }
                try { hold.DynamicInvoke(6); return -6; }
                catch (TargetInvocationException wrapped) when (wrapped.InnerException is Marker) { }
                Collect();
                return _held!.TryGetTarget(out _) ? -7 : 42;
            }
            case 3:
                try { HoldThenThrowPayload(8); return -8; }
                catch (PayloadException observed) when (FilterThrowsInternally(observed))
                {
                    Collect();
                    if (observed.Value.Number != 42) return -9;
                }
                return _held!.TryGetTarget(out _) ? -10 : 42;
            default:
                throw new ArgumentOutOfRangeException(nameof(input));
        }
    }

    public static int Run(int input)
    {
        _events = new int[6];
        _count = 0;
        LastDisposed = null;
        switch (input)
        {
            case 0:
                try
                {
                    try { Mark(1); ThrowPayload(); }
                    finally { Mark(3); GC.Collect(); }
                }
                catch (PayloadException observed) when (Inspect(observed))
                {
                    Mark(4);
                    GC.Collect();
                    if (observed.Value.Number != 42) return -1;
                }
                return Matches([1, 2, 3, 4]);
            case 1:
            {
                var value = new Payload(41);
                try
                {
                    try { Mark(1); ThrowMarker(); }
                    finally { Mark(2); GC.Collect(); value.Number++; }
                }
                catch (Marker) { Mark(3); GC.Collect(); }
                return value.Number == 42 ? Matches([1, 2, 3]) : -2;
            }
            case 2:
            {
                ref var interior = ref CreateInterior();
                try
                {
                    try { Mark(1); ThrowMarker(); }
                    finally { Mark(2); GC.Collect(); interior++; }
                }
                catch (Marker) { Mark(3); GC.Collect(); }
                return interior == 42 ? Matches([1, 2, 3]) : -3;
            }
            case 3:
                try
                {
                    using var resource = new Resource();
                    Mark(1);
                    ThrowMarker();
                }
                catch (Marker) { Mark(4); GC.Collect(); }
                return LastDisposed?.Number == 42 ? Matches([1, 2, 3, 4]) : -4;
            case 4:
                try
                {
                    try { Mark(1); ThrowPayload(); }
                    finally { Mark(4); GC.Collect(); }
                }
                catch (PayloadException observed) when (InspectAndThrow(observed)) { return -5; }
                catch (PayloadException observed)
                {
                    Mark(3);
                    GC.Collect();
                    if (observed.Value.Number != 42) return -6;
                }
                return Matches([1, 2, 4, 3]);
            case 5:
            {
                var original = CreateException();
                try
                {
                    try { Mark(1); throw original; }
                    catch (PayloadException observed)
                    {
                        Mark(2);
                        GC.Collect();
                        if (!ReferenceEquals(observed, original) || observed.Value.Number != 42) return -7;
                        throw;
                    }
                    finally { Mark(3); GC.Collect(); }
                }
                catch (PayloadException observed)
                {
                    Mark(4);
                    GC.Collect();
                    if (!ReferenceEquals(observed, original) || observed.Value.Number != 42) return -8;
                }
                return Matches([1, 2, 3, 4]);
            }
            default:
                throw new ArgumentOutOfRangeException(nameof(input));
        }
    }

    public static void Mark(int value) => _events[_count++] = value;

    private static int Matches(int[] expected)
    {
        if (_count != expected.Length) return -9;
        for (var index = 0; index < _count; index++)
            if (_events[index] != expected[index]) return -10;
        return 42;
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static PayloadException CreateException() => new(new Payload(42));

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static void ThrowPayload() => throw CreateException();

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static void ThrowMarker() => throw new Marker();

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static ref int CreateInterior() => ref new Payload(41).Number;

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static bool Inspect(PayloadException value)
    {
        Mark(2);
        GC.Collect();
        return value.Value.Number == 42;
    }

    private static void Collect()
    {
        GC.Collect();
        GC.Collect();
    }

    // Each frame keeps its payload live across the call that throws, so only the
    // frames the exception skipped still refer to the payload the weak reference names.
    [MethodImpl(MethodImplOptions.NoInlining)]
    private static void Hold(int depth)
    {
        var value = new Payload(depth);
        if (depth == 0) throw new Marker();
        if (depth == 1) _held = new WeakReference<Payload>(value);
        Hold(depth - 1);
        Sink(value);
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static void Sink(Payload value) => _sink += value.Number;

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static int HoldAndThrow(int depth)
    {
        Hold(depth);
        return 0;
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static void HoldThenThrowPayload(int depth)
    {
        var value = new Payload(depth);
        if (depth == 0) ThrowPayload();
        if (depth == 1) _held = new WeakReference<Payload>(value);
        HoldThenThrowPayload(depth - 1);
        Sink(value);
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static bool FilterThrowsInternally(PayloadException value)
    {
        var held = _held;
        try { Hold(10); }
        catch (Marker) { Collect(); }
        _held = held;
        return value.Value.Number == 42;
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static void Touch(ref Block block) => block.B = block.A + 1;

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static long Fill(int depth)
    {
        var block = new Block { A = depth, H = depth * 2 };
        Touch(ref block);
        if (depth == 0) throw new Marker();
        return Fill(depth - 1) + block.A + block.H + block.B - block.A - 1;
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static long Sum(int depth)
    {
        var block = new Block { A = depth, H = depth * 2 };
        Touch(ref block);
        var inner = depth == 0 ? 0 : Sum(depth - 1);
        return inner + block.A + block.H + block.B - block.A - 1;
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static bool InspectAndThrow(PayloadException value)
    {
        if (!Inspect(value)) return true;
        throw new Marker();
    }
}
